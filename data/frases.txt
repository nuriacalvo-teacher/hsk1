# Formato de cada tema:
#   # número | título en chino | título en español
# Formato de cada frase (separadores " | "):
#   hanzi | pinyin | español [ / otra traducción válida / … ] [| opciones]
# Opciones (opcionales, separadas por espacios en la última columna):
#   tts=texto     texto que se envía a la voz si el hanzi lleva cifras que se
#                 leerían mal (teléfonos, años)
#   py=otro       otro pinyin válido (para "español → pinyin")
#   zh=otro       otro hanzi válido (para "español → hanzi")
